import bpy
import os

# =========================
# OUTPUT PATHS
# =========================
output_dir = "C:/your/export/folder"  # change this
model_path = os.path.join(output_dir, "model_only.glb")
anim_path  = os.path.join(output_dir, "nla_animation.glb")

os.makedirs(output_dir, exist_ok=True)

# =========================
# 1. EXPORT MODEL ONLY
# =========================

bpy.ops.object.select_all(action='DESELECT')

# Select only mesh objects
for obj in bpy.data.objects:
    if obj.type == 'MESH':
        obj.select_set(True)

bpy.ops.export_scene.gltf(
    filepath=model_path,
    export_format='GLB',
    use_selection=True,
    export_animations=False,
    export_skins=False
)

print("Model exported:", model_path)

# =========================
# 2. EXPORT NLA ANIMATION
# =========================

bpy.ops.object.select_all(action='DESELECT')

# Select armature (needed for NLA animation)
armatures = [obj for obj in bpy.data.objects if obj.type == 'ARMATURE']
for obj in armatures:
    obj.select_set(True)

# Ensure NLA strips are evaluated
bpy.context.view_layer.update()

bpy.ops.export_scene.gltf(
    filepath=anim_path,
    export_format='GLB',

    use_selection=True,

    # IMPORTANT: enable animation export
    export_animations=True,
    export_skins=True,

    # IMPORTANT: this makes Blender bake NLA strips into glTF clips
    export_nla_strips=True,

    # safer for game engines
    export_apply=True
)

print("NLA animation exported:", anim_path)
